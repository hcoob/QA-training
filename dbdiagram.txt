Table Customers {
  customer_id integer [primary key]
  customer_forename varchar
  customer_surname varchar
  email varchar
}

Table Address {
  adress_id integer [primary key]
  customer_id integer
  adress varchar
}

Table Products {
  product_id integer [primary key]
  product_name varchar
  category_id integer 
}

Table Category {
  category_id integer [primary key]
  category_name varchar
}

Table Orders {
  order_id integer [primary key]
  customer_id integer 
  product_id integer
  notes text
  created_at timestamp
}

Ref Customers_Orders: Customers.customer_id > Orders.customer_id 
Ref Products_Orders: Products.product_id > Orders.product_id
Ref Category_Products: Category.category_id > Products.category_id
Ref Customers_Address: Customers.customer_id > Address.customer_id
